cmake_minimum_required(VERSION 3.16)
project(florent_rl CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(florent_rl_core STATIC
  sim/florent_actions.cpp
  sim/florent_heads.cpp
  sim/florent_game.cpp
  sim/map_gen.cpp
)
set_target_properties(florent_rl_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
# engine + heads C API in one library, for the Python parity tests
add_library(rlcapi SHARED sim/heads_capi.cpp ../engine/src/capi.cpp)
target_link_libraries(rlcapi PRIVATE florent_rl_core)
target_include_directories(florent_rl_core PUBLIC ../engine/src sim train)
target_sources(florent_rl_core PRIVATE ../engine/src/fengine.cpp ../engine/src/obs.cpp)
add_executable(bus_stats tests/bus_stats.cpp)
target_link_libraries(bus_stats PRIVATE florent_rl_core)

# LibTorch: either a standalone libtorch (find_package) or the pip torch wheel
# (set TORCH_PY=<site-packages>/torch; its CMake config demands CUDA, so link it by hand).
if(TORCH_PY)
  set(Torch_FOUND TRUE)
  set(TORCH_INCLUDE_DIRS ${TORCH_PY}/include ${TORCH_PY}/include/torch/csrc/api/include)
  set(TORCH_LIBRARIES ${TORCH_PY}/lib/libtorch.so ${TORCH_PY}/lib/libtorch_cpu.so ${TORCH_PY}/lib/libc10.so)
  # a CUDA build of libtorch: link its CUDA libraries too (kept even if no
  # symbol is referenced, so torch::cuda::is_available() sees them)
  if(EXISTS ${TORCH_PY}/lib/libtorch_cuda.so)
    set(TORCH_LIBRARIES -Wl,--no-as-needed ${TORCH_PY}/lib/libtorch_cuda.so ${TORCH_PY}/lib/libc10_cuda.so -Wl,--as-needed ${TORCH_LIBRARIES})
    message(STATUS "libtorch: CUDA build")
  endif()
  set(CMAKE_BUILD_RPATH ${TORCH_PY}/lib)
else()
  find_package(Torch QUIET)
endif()
if(NOT Torch_FOUND)
  message(WARNING "LibTorch not found: neural_train / distill are not built. Pass -DTORCH_PY=/path/to/libtorch (or the pip torch folder).")
endif()
find_package(OpenMP REQUIRED)
if(Torch_FOUND)
 # The fused CUDA actor trunk (train/trunk_kernels.cu) when a CUDA compiler
 # and a CUDA libtorch are there; otherwise a stub (PyTorch ops only).
 include(CheckLanguage)
 check_language(CUDA)
 set(_fused OFF)
 if(CMAKE_CUDA_COMPILER AND (EXISTS ${TORCH_PY}/lib/libtorch_cuda.so OR TARGET torch_cuda))
   enable_language(CUDA)
   find_package(CUDAToolkit)
   if(CUDAToolkit_FOUND)
     set(_fused ON)
   endif()
 endif()
 if(_fused)
   if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES OR CMAKE_CUDA_ARCHITECTURES STREQUAL "52")
     set(CMAKE_CUDA_ARCHITECTURES native)
   endif()
   add_library(florent_trunk STATIC train/trunk_kernels.cu train/critic_kernels.cu train/trunk_fn.cpp)
   set_target_properties(florent_trunk PROPERTIES CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}" POSITION_INDEPENDENT_CODE ON)
   target_compile_options(florent_trunk PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
   target_include_directories(florent_trunk PRIVATE train ${TORCH_INCLUDE_DIRS} ${CUDAToolkit_INCLUDE_DIRS})
   target_compile_definitions(florent_trunk PRIVATE $<$<COMPILE_LANGUAGE:CXX>:_GLIBCXX_USE_CXX11_ABI=1>)
   target_link_libraries(florent_trunk PUBLIC ${TORCH_LIBRARIES} CUDA::cudart)
   message(STATUS "fused CUDA actor trunk: on (arch ${CMAKE_CUDA_ARCHITECTURES})")
 else()
   add_library(florent_trunk STATIC train/trunk_stub.cpp)
   target_include_directories(florent_trunk PRIVATE train ${TORCH_INCLUDE_DIRS})
   target_compile_definitions(florent_trunk PRIVATE _GLIBCXX_USE_CXX11_ABI=1)
   target_link_libraries(florent_trunk PUBLIC ${TORCH_LIBRARIES})
   message(STATUS "fused CUDA actor trunk: off (no CUDA compiler / CUDA libtorch)")
 endif()
 add_library(florent_model INTERFACE)
 target_include_directories(florent_model INTERFACE train ${TORCH_INCLUDE_DIRS})
 target_link_libraries(florent_model INTERFACE florent_trunk ${TORCH_LIBRARIES})
 target_compile_definitions(florent_model INTERFACE _GLIBCXX_USE_CXX11_ABI=1)
endif()
add_executable(florent_train train/main.cpp)
target_include_directories(florent_train PRIVATE train sim)
target_link_libraries(florent_train PRIVATE florent_rl_core)
if(Torch_FOUND)
 add_executable(neural_train train/neural_train.cpp)
 target_include_directories(neural_train PRIVATE train sim ${TORCH_INCLUDE_DIRS})
 target_link_libraries(neural_train PRIVATE florent_rl_core florent_model ${TORCH_LIBRARIES} OpenMP::OpenMP_CXX)
 # CUDA graphs for the rollout forward: needs a CUDA libtorch and the CUDA
 # runtime headers (a CUDA toolkit, or the pip nvidia-*-runtime package).
 set(_cuda_lib "")
 if(TORCH_PY AND EXISTS ${TORCH_PY}/lib/libtorch_cuda.so)
   set(_cuda_lib ON)
 elseif(TARGET torch_cuda OR TORCH_CUDA_LIBRARIES)
   set(_cuda_lib ON)
 endif()
 if(_cuda_lib)
   file(GLOB _pip_cuda_inc ${TORCH_PY}/../nvidia/*/include ${TORCH_PY}/../nvidia/cuda_runtime/include)
   find_path(CUDA_RUNTIME_INC cuda_runtime_api.h HINTS ${_pip_cuda_inc} $ENV{CUDA_HOME}/include /usr/local/cuda/include /usr/include)
   file(GLOB _pip_cuda_lib ${TORCH_PY}/../nvidia/*/lib ${TORCH_PY}/../nvidia/cuda_runtime/lib)
   find_library(CUDART_LIB NAMES cudart libcudart.so.13 libcudart.so.12 HINTS ${_pip_cuda_lib} $ENV{CUDA_HOME}/lib64 /usr/local/cuda/lib64)
   if(CUDA_RUNTIME_INC AND CUDART_LIB)
     target_include_directories(neural_train PRIVATE ${CUDA_RUNTIME_INC})
     target_link_libraries(neural_train PRIVATE ${CUDART_LIB})
     target_compile_definitions(neural_train PRIVATE FLORENT_CUDA_GRAPHS)
     message(STATUS "neural_train: rollout CUDA graphs on (cuda_runtime_api.h in ${CUDA_RUNTIME_INC})")
   else()
     message(STATUS "neural_train: CUDA runtime headers / libcudart not found, rollout CUDA graphs off")
   endif()
 endif()
 target_compile_definitions(neural_train PRIVATE _GLIBCXX_USE_CXX11_ABI=1)
 # Spatial checkpoint export is implemented by rl/bot/export.py after a
 # tensor-to-JSON conversion; no stale dense-MLP exporter target is shipped.
endif()
